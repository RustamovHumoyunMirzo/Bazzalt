cmake_minimum_required(VERSION 3.20)
project(BazzaltJoltDependency LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Match the game SDK ABI in the dependency-only CI probe.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "The Linux SDK probe requires Clang")
    endif()
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
        set(_stdlib libstdc++)
    else()
        set(_stdlib libc++)
    endif()
    add_compile_options(-stdlib=${_stdlib})
    add_link_options(-stdlib=${_stdlib})
endif()
# Deliberately use opposite defaults to detect upstream option/flag leakage.
set(BUILD_SHARED_LIBS ON)
set(_caller_flags "${CMAKE_CXX_FLAGS}")
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/Jolt.cmake")
if(NOT BUILD_SHARED_LIBS OR NOT CMAKE_CXX_STANDARD EQUAL 20 OR NOT CMAKE_CXX_FLAGS STREQUAL _caller_flags)
    message(FATAL_ERROR "Jolt dependency changed caller build settings")
endif()
if(BAZZALT_BUILD_JOLT)
    get_target_property(_jolt_type Jolt TYPE)
    get_target_property(_jolt_pic Jolt POSITION_INDEPENDENT_CODE)
    if(NOT _jolt_type STREQUAL "STATIC_LIBRARY" OR NOT _jolt_pic)
        message(FATAL_ERROR "The private Jolt dependency must be static PIC")
    endif()
    add_executable(BazzaltJoltDependencyProbe DependencyProbe.cpp)
    target_link_libraries(BazzaltJoltDependencyProbe PRIVATE Bazzalt::JoltDependency)
    enable_testing()
    add_test(NAME BazzaltJoltDependencyProbe COMMAND BazzaltJoltDependencyProbe)
endif()
