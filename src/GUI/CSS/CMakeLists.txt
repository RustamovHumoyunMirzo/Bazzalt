cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(BazzaltCSS LANGUAGES CXX)
    include(CTest)
endif()
add_library(BazzaltCSS STATIC Lexer.cpp Parser.cpp Selectors.cpp Values.cpp Cascade.cpp Requirements.cpp Motion.cpp)
target_compile_features(BazzaltCSS PUBLIC cxx_std_20)
target_include_directories(BazzaltCSS PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set_target_properties(BazzaltCSS PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(BazzaltCSS PRIVATE /utf-8 /W4)
else()
    target_compile_options(BazzaltCSS PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND BUILD_TESTING)
    add_executable(BazzaltCSSTests ../../../tests/CSSTests.cpp)
    target_link_libraries(BazzaltCSSTests PRIVATE BazzaltCSS)
    add_test(NAME BazzaltCSSTests COMMAND BazzaltCSSTests)
    set_tests_properties(BazzaltCSSTests PROPERTIES TIMEOUT 20)
endif()
